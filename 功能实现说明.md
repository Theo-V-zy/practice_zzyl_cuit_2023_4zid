```java
@RequestMapping("/sysMenus")
public List<Menu> sysMenusList(HttpSession session){
    List<Menu> allMenus = menuService.querySysMenuList();
    Object online = session.getAttribute("online");
    if (online == null) { return allMenus; }

    UserLineDto user = (UserLineDto) online;
    String menuIds = user.getMenuIds(); // 从 session 取

    // 管理员有全部菜单，跳过过滤
    if (menuIds == null || menuIds.isEmpty() || menuIds.contains("1,2,...,24")) {
        return allMenus;
    }

    // 按逗号拆分，过滤菜单树
    Set<Integer> allowedIds = Arrays.stream(menuIds.split(","))
        .map(String::trim).filter(s -> !s.isEmpty())
        .map(Integer::parseInt).collect(Collectors.toSet());
    return filterMenus(allMenus, allowedIds);
}
```

### 1.3 过滤算法：保留节点 + 祖先链
`backend/src/main/java/com/soft/controller/MenuController.java` 第 48-72 行

```java
private List<Menu> filterMenus(List<Menu> allMenus, Set<Integer> allowedIds) {
    Map<Integer, Menu> flatMap = new HashMap<>();
    flattenTree(allMenus, flatMap);              // 1. 拍平树

    Set<Integer> keepIds = new HashSet<>(allowedIds);
    for (Integer id : allowedIds) {               // 2. 补祖先节点
        Menu m = flatMap.get(id);
        while (m != null && m.getPid() != null && m.getPid() != 0) {
            keepIds.add(m.getPid());
            m = flatMap.get(m.getPid());
        }
    }

    List<Menu> filtered = new ArrayList<>();       // 3. 重建过滤树
    for (Menu menu : allMenus) {
        if (keepIds.contains(menu.getId())) {
            Menu copy = copyMenu(menu);
            copy.setSubItems(filterChildrenFlat(allMenus, keepIds, menu.getId()));
            filtered.add(copy);
        }
    }
    return filtered;
}
```

### 2.3 OSS 上传工具

```java
public boolean isConfigured() {
    return hasText(System.getenv("OSS_ACCESS_KEY_ID"))
        && hasText(System.getenv("OSS_ACCESS_KEY_SECRET"));
}

public String uploadFile(String objectName, byte[] content) {
    OSS ossClient = null;
    try {
        EnvironmentVariableCredentialsProvider credentialsProvider
            = CredentialsProviderFactory.newEnvironmentVariableCredentialsProvider();
        ClientBuilderConfiguration config = new ClientBuilderConfiguration();
        config.setSignatureVersion(SignVersion.V4);
        ossClient = OSSClientBuilder.create()
            .endpoint(endpoint).credentialsProvider(credentialsProvider)
            .clientConfiguration(config).region(region).build();
        ossClient.putObject(bucketName, objectName, new ByteArrayInputStream(content));
    } catch (Exception ex) {
        throw new IllegalStateException("OSS upload failed", ex);
    } finally {
        if (ossClient != null) ossClient.shutdown();
    }
    // 拼 OSS URL: https://bucket.endpoint/objectName
    return endpoint.split("//")[0] + "//" + bucketName + "."
        + endpoint.split("//")[1] + "/" + objectName;
}
```

### 4.1 后端 SSE 流式接口
`backend/src/main/java/com/soft/controller/AIChatController.java`

```java
@PostMapping("/api/chat/stream")
public SseEmitter chatStream(@RequestBody Map<String, Object> body) {
    String userMessage = (String) body.get("message");
    SseEmitter emitter = new SseEmitter(120000L);

    new Thread(() -> {
        HttpURLConnection conn = null;
        try {
            // 构建 messages: system prompt + 历史 + 当前
            String jsonBody = buildRequestBody(userMessage, history);

            conn = (HttpURLConnection) new URL(apiUrl + "/chat/completions").openConnection();
            conn.setRequestMethod("POST");
            conn.setRequestProperty("Authorization", "Bearer " + apiKey);
            conn.setRequestProperty("Content-Type", "application/json");
            conn.setDoOutput(true);
            // 发送请求
            conn.getOutputStream().write(jsonBody.getBytes(StandardCharsets.UTF_8));

            // 逐行读取 SSE 流
            BufferedReader reader = new BufferedReader(
                new InputStreamReader(conn.getInputStream()));
            String line;
            while ((line = reader.readLine()) != null) {
                if (line.startsWith("data: ")) {
                    String data = line.substring(6);
                    if ("[DONE]".equals(data)) break;
                    String content = extractContent(data);  // 从 JSON 提取 delta.content
                    if (content != null && !content.isEmpty()) {
                        emitter.send(SseEmitter.event().data(content)); // 逐字推送
                    }
                }
            }
            emitter.complete();
        } catch (Exception e) {
            emitter.completeWithError(e);
        }
    }).start();
    return emitter;
}
```

### 4.2 前端 SSE 消费
`frontend/src/views/admin/AIChat.vue` 第 88-120 行

```javascript
async function sendMessage() {
  const text = inputText.value.trim()
  if (!text || streaming.value) return
  messages.value.push({ role: 'user', content: text })

  streaming.value = true
  const res = await fetch(BASE_URL + '/api/chat/stream', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ message: text, history })
  })

  const reader = res.body.getReader()     // 读流
  const decoder = new TextDecoder()
  let buffer = ''
  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })
    const lines = buffer.split('\n')
    buffer = lines.pop() || ''
    for (const line of lines) {
      if (line.startsWith('data:')) {
        streamingText.value += line.slice(5).trim()  // 追显
      }
    }
  }
  // 完成后保存到消息列表
  if (streamingText.value) {
    messages.value.push({ role: 'assistant', content: streamingText.value })
  }
  streaming.value = false
}
```

```java
public Map<String, Object> page(@RequestBody MessagePageDto dto, HttpSession session) {
    Object online = session.getAttribute("online");
    if (online == null) { result.put("code", 401); result.put("msg", "请先登录"); return result; }
    UserLineDto user = (UserLineDto) online;

    QueryWrapper<Message> wrapper = new QueryWrapper<>();
    wrapper.eq("receiver_type", "USER").eq("receiver_id", user.getId())
           .eq("enabled", 1);
    if (dto.getReadStatus() != null) wrapper.eq("read_status", dto.getReadStatus());
    wrapper.orderByDesc("create_time");

    Page<Message> page = messageService.page(
        new Page<>(dto.getPage(), dto.getPageSize()), wrapper);
    result.put("code", 200);
    result.put("data", page.getRecords());
    result.put("total", page.getTotal());
    return result;
}
```

```javascript
function request(options) {
  return new Promise((resolve, reject) => {
    uni.request({
      url: 'http://localhost:8080' + options.url,
      method: options.method || 'GET',
      data: options.data || {},
      header: { 'Content-Type': 'application/json' },
      success(res) {
        if (res.statusCode === 200 && res.data.code === 200) resolve(res.data)
        else if (res.data.code === 401) {
          uni.reLaunch({ url: '/pages/login/login' })  // session 过期→登录页
        } else {
          uni.showToast({ title: res.data.msg || '请求失败', icon: 'none' })
        }
      }
    })
  })
}
```
